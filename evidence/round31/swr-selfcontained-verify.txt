#!/bin/bash
# Savage Wind (BOTW 1.9.0) native x86-64 launcher for Steam Deck.
# Add THIS FILE to Steam as a non-Steam game.
# Game files: put extracted/ (exefs + romfs) in ~/.local/share/savagewind/extracted/
# or set SAVAGEWIND_DATA_ROOT=/path/to/folder
set -u
here="$(cd "$(dirname "$(readlink -f "${BASH_SOURCE[0]}")")" && pwd)"
export LD_LIBRARY_PATH="$here/usr/lib${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}"
exec "$here/usr/bin/savagewind" "$@"
